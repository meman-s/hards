## 1. Обход неориентированного графа в ширину

### Ответ 1.1

```python
from collections import deque
from typing import List, Optional

def bfs_undirected_1(
    graph: List[List[int]],
    start: int,
    n: Optional[int] = None,
) -> List[int]:
    n = n if n is not None else len(graph)
    order: List[int] = []
    visited = [False] * n
    queue: deque[int] = deque([start])
    visited[start] = True

    while queue:
        v = queue.popleft()
        order.append(v)
        for u in graph[v]:
            if not visited[u]:
                visited[u] = True
                queue.append(u)

    return order
```

**Объяснение:** Используется очередь (deque) для обхода графа в ширину. Вершины посещаются по уровням: сначала стартовая, затем все её соседи, затем соседи соседей и т.д.

---

### Ответ 1.2

```python
    n: Optional[int] = None,

    distances = [-1] * n

    visited[start] = True
    distances[start] = 0

    while queue:
        v = queue.popleft()
        for u in graph[v]:
            if not visited[u]:
                visited[u] = True
                distances[u] = distances[v] + 1
                queue.append(u)

```

**Объяснение:** При обходе в ширину расстояние до каждой новой вершины равно расстоянию до текущей вершины плюс 1. Недостижимые вершины остаются с расстоянием -1.

```python
from typing import List, Optional

def dfs_undirected_2(
    graph: List[List[int]],
    n: Optional[int] = None,
) -> bool:
    n = n if n is not None else len(graph)
    if n == 0:
        return True
    
    visited = [False] * n

    def dfs(v: int) -> None:
        visited[v] = True
        for u in graph[v]:
            if not visited[u]:
                dfs(u)

    dfs(0)
    return all(visited)
```

**Объяснение:** Граф связный, если из любой вершины можно достичь всех остальных. Начинаем обход из вершины 0 и проверяем, что все вершины были посещены.

### Ответ 3.2

```python
from collections import deque
from typing import List, Optional, Set

def bfs_directed_2(
    graph: List[List[int]],
    start: int,
    n: Optional[int] = None,
) -> Set[int]:
    n = n if n is not None else len(graph)
    visited = [False] * n
    queue: deque[int] = deque([start])
    visited[start] = True
    reachable: Set[int] = {start}

    while queue:
        v = queue.popleft()
        for u in graph[v]:
            if not visited[u]:
                visited[u] = True
                reachable.add(u)
                queue.append(u)

    return reachable
```

**Объяснение:** Используем BFS для поиска всех вершин, достижимых из стартовой. Результат возвращаем в виде множества для удобства.

```python
from typing import List, Optional

def dfs_directed_2(
    graph: List[List[int]],
    n: Optional[int] = None,
) -> List[List[int]]:
    n = n if n is not None else len(graph)
    
    def dfs(v: int, visited: List[bool], component: List[int]) -> None:
        visited[v] = True
        component.append(v)
        for u in graph[v]:
            if not visited[u]:
                dfs(u, visited, component)
    
    def get_transpose(g: List[List[int]], size: int) -> List[List[int]]:
        transpose: List[List[int]] = [[] for _ in range(size)]
        for i in range(size):
            for j in g[i]:
                transpose[j].append(i)
        return transpose
    
    visited = [False] * n
    order: List[int] = []
    
    for i in range(n):
        if not visited[i]:
            dfs(i, visited, order)
    
    transpose = get_transpose(graph, n)
    visited = [False] * n
    components: List[List[int]] = []
    
    for i in reversed(order):
        if not visited[i]:
            component: List[int] = []
            dfs(i, visited, component)
            components.append(component)
    
    return components
```

**Объяснение:** Используется алгоритм Косарайю. Сначала делаем DFS на исходном графе и сохраняем порядок завершения. Затем делаем DFS на транспонированном графе в обратном порядке - каждая компонента связности будет найдена отдельно.

## 5. Топологическая сортировка

### Ответ 5.1

```python
from typing import List, Optional

def topological_sort_1(
    graph: List[List[int]],
    n: int,
) -> Optional[List[int]]:
    WHITE, GRAY, BLACK = 0, 1, 2
    color = [WHITE] * n
    result: List[int] = []

    def dfs(v: int) -> bool:
        color[v] = GRAY
        for u in graph[v]:
            if color[u] == GRAY:
                return False
            if color[u] == WHITE and not dfs(u):
                return False
        color[v] = BLACK
        result.append(v)
        return True

    for start in range(n):
        if color[start] == WHITE and not dfs(start):
            return None

    return result[::-1]
```

**Объяснение:** Используется DFS с цветовой маркировкой. GRAY означает вершину в процессе обработки, BLACK - обработанную. Если встречаем GRAY вершину, значит есть цикл. Результат разворачиваем, так как вершины добавляются в обратном порядке.

---

### Ответ 5.2

```python
from typing import List, Set
from collections import defaultdict

def topological_sort_2(
    graph: List[List[int]],
    n: int,
) -> List[List[int]]:
    in_degree = [0] * n
    for i in range(n):
        for j in graph[i]:
            in_degree[j] += 1
    
    def has_cycle() -> bool:
        color = [0] * n
        def dfs(v: int) -> bool:
            color[v] = 1
            for u in graph[v]:
                if color[u] == 1:
                    return True
                if color[u] == 0 and dfs(u):
                    return True
            color[v] = 2
            return False
        for i in range(n):
            if color[i] == 0 and dfs(i):
                return True
        return False
    
    if has_cycle():
        return []
    
    def all_topological_sorts() -> List[List[int]]:
        result: List[List[int]] = []
        visited = [False] * n
        current: List[int] = []
        
        def backtrack():
            if len(current) == n:
                result.append(current[:])
                return
            
            for v in range(n):
                if not visited[v] and in_degree[v] == 0:
                    visited[v] = True
                    current.append(v)
                    for u in graph[v]:
                        in_degree[u] -= 1
                    
                    backtrack()
                    
                    for u in graph[v]:
                        in_degree[u] += 1
                    current.pop()
                    visited[v] = False
        
        backtrack()
        return result
    
    return all_topological_sorts()
```

**Объяснение:** Сначала проверяем наличие цикла. Если цикла нет, используем backtracking для генерации всех возможных топологических сортировок. На каждом шаге выбираем вершину с нулевой входящей степенью, обрабатываем её и рекурсивно продолжаем.

```python
import pstats

def find_duplicates(items):
    duplicates = []
    for i in range(len(items)):
        for j in range(i + 1, len(items)):
            if items[i] == items[j]:
                duplicates.append(items[i])

    for item in data:
        if item not in result:
            result.append(item)
```

**Узкие места:**
1. В `find_duplicates`: O(n²) сложность из-за вложенных циклов
2. В `process_large_dataset`: O(n²) сложность из-за проверки `item not in result` (линейный поиск в списке)

**Оптимизированные версии:**
```python
def find_duplicates_optimized(items):
    seen = set()
    duplicates = set()
    for item in items:
        if item in seen:
            duplicates.add(item)
        else:
            seen.add(item)
    return list(duplicates)

def process_large_dataset_optimized(data):
    seen = set()
    result = []
    for item in data:
        if item not in seen:
            seen.add(item)
            result.append(item)
    return result
```

**Объяснение:**
1. `find_duplicates_optimized`: Использует set для O(1) проверки наличия элемента, сложность O(n)
2. `process_large_dataset_optimized`: Использует set для отслеживания уже добавленных элементов, сложность O(n) вместо O(n²)

```python
def count_pairs_with_sum_optimized(arr, target):
    from collections import Counter
    freq = Counter(arr)
    count = 0
    for x in freq:
        y = target - x
        if y < x:
            continue
        if y == x:
            count += freq[x] * (freq[x] - 1) // 2
        elif y in freq:
            count += freq[x] * freq[y]
```

```sql
    department_id INTEGER PRIMARY KEY REFERENCES departments(id),
    employee_count INTEGER DEFAULT 0,
    total_salary DECIMAL(12, 2) DEFAULT 0
);

CREATE OR REPLACE FUNCTION update_department_stats()
RETURNS TRIGGER AS $$
BEGIN
    IF TG_OP = 'INSERT' THEN
        INSERT INTO department_stats (department_id, employee_count, total_salary)
        VALUES (NEW.department_id, 1, NEW.salary)
        ON CONFLICT (department_id) 
        DO UPDATE SET 
            employee_count = department_stats.employee_count + 1,
            total_salary = department_stats.total_salary + NEW.salary;
        RETURN NEW;
    ELSIF TG_OP = 'UPDATE' THEN
        UPDATE department_stats
        SET total_salary = total_salary - OLD.salary + NEW.salary
        WHERE department_id = NEW.department_id;
        RETURN NEW;
    ELSIF TG_OP = 'DELETE' THEN
        UPDATE department_stats
        SET 
            employee_count = employee_count - 1,
            total_salary = total_salary - OLD.salary
        WHERE department_id = OLD.department_id;
        RETURN OLD;
    END IF;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trigger_update_department_stats
AFTER INSERT OR UPDATE OR DELETE ON employees
FOR EACH ROW EXECUTE FUNCTION update_department_stats();
```

```python
from typing import List, Optional

def shortest_path_undirected(
    graph: List[List[int]],
    start: int,
    end: int,
    n: Optional[int] = None,
) -> int:
    n = n if n is not None else len(graph)
    if start == end:
        return 0
    dist = [-1] * n
    queue: deque[int] = deque([start])
    dist[start] = 0
    while queue:
        v = queue.popleft()
        for u in graph[v]:
            if dist[u] == -1:
                dist[u] = dist[v] + 1
                if u == end:
                    return dist[u]
                queue.append(u)
    return -1
```

```python
def longest_path_dag(
    graph: List[List[int]],
    n: int,
) -> Optional[int]:
    WHITE, GRAY, BLACK = 0, 1, 2
    color = [WHITE] * n
    order: List[int] = []

    def dfs(v: int) -> bool:
        color[v] = GRAY
        for u in graph[v]:
            if color[u] == GRAY:
                return False
            if color[u] == WHITE and not dfs(u):
                return False
        color[v] = BLACK
        order.append(v)
        return True

        if color[v] == WHITE and not dfs(v):
            return None
    order.reverse()
    dist = [0] * n
    for v in order:
        for u in graph[v]:
            dist[u] = max(dist[u], dist[v] + 1)
    return max(dist) if n else 0
```

```python
from typing import List, Optional, Tuple

def sources_and_sinks(
    graph: List[List[int]],
    n: Optional[int] = None,
) -> Tuple[List[int], List[int]]:
    n = n if n is not None else len(graph)
    in_deg = [0] * n
    for v in range(n):
        for u in graph[v]:
            in_deg[u] += 1
    sources = [v for v in range(n) if in_deg[v] == 0]
    sinks = [v for v in range(n) if len(graph[v]) == 0]
    return (sources, sinks)
```
